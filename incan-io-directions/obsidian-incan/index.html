def checked(n: int) -> Result[int, str]:
    if n <= 0:
        return Err("Must be positive")
    return Ok(n)
 
def main() -> None:
    match checked(42):
        case Ok(value):
            println(value)
        case Err(message):
            println(message)

combinators.incn
def positive(score: int) -> bool:
    return score > 0
 
def double(score: int) -> int:
    return score * 2
 
def main() -> None:
    ready = [3, -1, 5, 8].iter()
        .filter(positive)
        .map(double)
        .take(2)
        .collect()
    println(ready[0])
